cmake_minimum_required(VERSION 3.25)
project(FinalGIS VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network)
find_package(GDAL 3.10 CONFIG REQUIRED)
find_path(FINALGIS_PROJ_DATA_DIR proj.db PATH_SUFFIXES share/proj REQUIRED)
include(CTest)
option(FINALGIS_SANITIZERS "Address and undefined behavior sanitizers" OFF)
add_library(finalgis_geo SHARED src/geography.cpp src/model.cpp src/workflow.cpp)
target_include_directories(finalgis_geo PUBLIC src)
target_link_libraries(finalgis_geo PUBLIC Qt6::Core Qt6::Network GDAL::GDAL)
target_compile_definitions(finalgis_geo PRIVATE FINALGIS_PROJ_DATA_DIR="${FINALGIS_PROJ_DATA_DIR}")
target_compile_options(finalgis_geo PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(FINALGIS_SANITIZERS)
 target_compile_options(finalgis_geo PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(finalgis_geo PUBLIC -fsanitize=address,undefined)
endif()
add_executable(finalgis-worker src/main.cpp)
target_link_libraries(finalgis-worker PRIVATE finalgis_geo)
if(BUILD_TESTING)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME finalgis_workflows COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/workflows.py $<TARGET_FILE:finalgis-worker> ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/montreal)
 add_executable(finalgis-c17-check tests/c_transform.c)
 set_target_properties(finalgis-c17-check PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED ON)
 target_link_libraries(finalgis-c17-check PRIVATE finalgis_geo)
 add_test(NAME finalgis_c17_abi COMMAND finalgis-c17-check)
 add_test(NAME finalgis_workflow_v2 COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/workflow_v2.py $<TARGET_FILE:finalgis-worker> ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/montreal)
 add_test(NAME finalgis_spatialdb_bridge COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/readback_bridge.py)
endif()
install(TARGETS finalgis_geo finalgis-worker LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES src/finalgis.h DESTINATION include)
install(DIRECTORY plugin/ DESTINATION share/finalgis/plugin PATTERN __pycache__ EXCLUDE)
